// Sharp chevron (contract, "Icons"): Hugeicons draws its chevrons as a curve with a round tip; this is the same
// 12 x 6 angle in straight lines, mitred at the tip like the tick. The one drawn path of plush, the same text in every file.
const sharpChevron = (d: string): IconSvgElement => [["path", { d, stroke: "currentColor", strokeLinecap: "round", strokeLinejoin: "miter", strokeWidth: 1.5, key: "0" }]]
const ChevronDownSharpIcon = sharpChevron("M6 9L12 15L18 9")

// The answer opens and closes over its measured height (plush-accordion-down/up on the calm spring), so the card grows and
// shrinks with it; the chevron turns on the spring; on closing the answer fades and drifts 10 px up, on opening it fades
// in one beat (--plush-beat) after the height starts. Switching questions is one move: the old answer closes and the new
// one opens at the same moment, on the same spring, so the card's height goes straight from the old to the new value, and
// the two texts cross-fade (no beat). A tap shows a grey pill under the question. The trigger keeps its 4 px margin below,
// so its focus ring never touches the answer.

// Same href and text as in progress-ring.tsx, so React keeps one.
// Registers --plush-pop-armed as a number (a 0s transition with a delay flips it, the pop keyframes read it). A React <style> with an href lands in <head> once per page (also in
// the server HTML), however often it renders. Not in tokens.css: the shadcn CLI writes @property into globals.css again on
// every add.
const popArmedProperty = (
  <style href="plush-property-pop-armed" precedence="plush">
    {'@property --plush-pop-armed { syntax: "<number>"; inherits: false; initial-value: 0; }'}
  </style>
)

function Accordion({
  className,
  ...props
}: React.ComponentProps<typeof AccordionPrimitive.Root>) {
  return (
    <>
      {popArmedProperty}
      <AccordionPrimitive.Root
        data-slot="accordion"
        className={cn("plush-font flex w-full max-w-full min-w-0 flex-col", className)}
        {...props}
      />
    </>
  )
}

function AccordionItem({
  className,
  ...props
}: React.ComponentProps<typeof AccordionPrimitive.Item>) {
  return (
    <AccordionPrimitive.Item
      data-slot="accordion-item"
      className={cn(
        // The separator: a 1 px line drawn inside the top of the next row (inset shadow), so it adds no height.
        "group/accordion-item mx-2.5 min-w-0 not-first:shadow-[inset_0_1px_0_var(--plush-fill)]",
        // Switching is one move: the opening item does not wait for a closing sibling (--plush-accordion-wait stays 0, read by
        // the height, the chevron and the gap below the question). Only its text: alone it fades in a beat after the height
        // starts; while a sibling's answer is still closing it fades in at once, so the two texts cross-fade instead of
        // leaving an empty answer between them.
        "[--plush-accordion-wait:0s] [--plush-accordion-fade-wait:var(--plush-beat)] [:has(>[data-state=closed]>[data-slot=accordion-content]:not([hidden]))>&[data-state=open]]:[--plush-accordion-fade-wait:0s]",
        // Armed: 1 while closed, 0 once open. An item that loads open is 0 from the start, so its answer is
        // simply there; one that opens stays 1 until its answer is fully in. The answer's keyframes (plush-accordion-down,
        // plush-accordion-fade-in) start from their end state times (1 - armed), so on load they run from end to end.
        "[--plush-pop-armed:1] data-[state=open]:[--plush-pop-armed:0] data-[state=open]:[transition:--plush-pop-armed_0s_linear_calc(var(--plush-accordion-wait)_+_var(--plush-beat)_+_max(var(--plush-dur-calm),var(--plush-dur-fade)))]",
        className
      )}
      {...props}
    />
  )
}

import * as React from "react"

// Three sizes: sm 28 (chat), default 32 (list rows), lg 46 (the stack). Letters are white and
// semibold on a colour: the fallback is the black accent unless a colour is passed (bg-(--plush-pink) …).
// In a group every circle wears a ring in the colour of the surface behind it (--plush-avatar-ring-color on the group or any
// parent, default --plush-bg) and overlaps its neighbour; with more than four circles and the plus (six children on) the stack huddles closer, on the soft spring.
// A circle that joins a stack in use (hovered, or focus inside it, e.g. on its plus button) comes out of its right neighbour:
// it starts there at --plush-pop-scale and grows in on the spring while the neighbour slides aside to make room, and its fallback
// colours in from white. The gate keeps a stack still on page load (@starting-style fires on first paint as well).

// Circle size and letter size of each step. The group sets them from the avatars it holds, so its count matches them.
const sizes = {
  self: "[--plush-avatar-size:var(--plush-avatar)] [--plush-avatar-text:var(--plush-text-md)] data-[size=sm]:[--plush-avatar-size:var(--plush-avatar-sm)] data-[size=sm]:[--plush-avatar-text:var(--plush-text-xs)] data-[size=lg]:[--plush-avatar-size:var(--plush-avatar-lg)] data-[size=lg]:[--plush-avatar-text:var(--plush-text-avatar-lg)]",
  group:
    "[--plush-avatar-size:var(--plush-avatar)] [--plush-avatar-text:var(--plush-text-md)] has-data-[size=sm]:[--plush-avatar-size:var(--plush-avatar-sm)] has-data-[size=sm]:[--plush-avatar-text:var(--plush-text-xs)] has-data-[size=lg]:[--plush-avatar-size:var(--plush-avatar-lg)] has-data-[size=lg]:[--plush-avatar-text:var(--plush-text-avatar-lg)]",
}

  ...props
}: React.ComponentProps<typeof AvatarPrimitive.Root> & {

}) {
  return (
    <AvatarPrimitive.Root
      data-slot="avatar"
      data-size={size}

        sizes.self,
        // An avatar that is itself a link (asChild on an <a href>) grows a little under the pointer, on the spring.
        "[transition:scale_var(--plush-dur-spring)_var(--plush-ease-spring)] [a[href]&]:hover:scale-(--plush-hover-grow)",
        // In a group: the ring, and springs for making room (margin), growing in and ducking (scale) and fading.
        "in-data-[slot=avatar-group]:ring-[length:var(--plush-avatar-ring)] in-data-[slot=avatar-group]:ring-[color:var(--plush-avatar-ring-color,var(--plush-bg))]",

        // Entry: lies on its right neighbour (margin end = minus one step) at the pop scale, then makes room and grows.
        // Gated on the group being hovered or holding focus, i.e. a stack in use.
        "starting:in-[[data-slot=avatar-group]:is(:hover,:focus-within)]:me-[calc(var(--plush-avatar-overlap-now)_-_var(--plush-avatar-size))]",
        "starting:in-[[data-slot=avatar-group]:is(:hover,:focus-within)]:scale-(--plush-pop-scale)",

  max: number

const CounterContext = React.createContext<CounterContextValue | null>(null)

function useCounter(part: string) {

const decimals = (n: number) => {
  const s = String(n)
  return s.includes(".") ? s.length - s.indexOf(".") - 1 : 0
}

function Counter({
  className,
  children,
  value: valueProp,
  defaultValue = 0,
  onValueChange,
  min = -Infinity,
  max = Infinity,
  step = 1,
  largeStep = step * 10,
  disabled = false,
  locale = "en-US",
  formatOptions,
  size = "default",
  variant = "default",
  name,
  ...props
}: Omit<React.ComponentProps<"div">, "defaultValue"> & {
  value?: number
  defaultValue?: number
  onValueChange?: (value: number) => void
  min?: number
  max?: number
  step?: number
  /** Page Up / Page Down. */
  largeStep?: number
  disabled?: boolean
  /** A fixed locale keeps the server and the browser on the same text. */
  locale?: string
  formatOptions?: Intl.NumberFormatOptions
  size?: "default" | "lg"
  variant?: "default" | "secondary"
  /** Submitted with a form under this name (a hidden input); the form's reset brings back defaultValue. */
  name?: string
}) {
  const [inner, setInner] = React.useState(defaultValue)
  const value = valueProp ?? inner
  const format = React.useMemo(() => new Intl.NumberFormat(locale, formatOptions), [locale, formatOptions])

  const setValue = (next: number) => {
    const p = 10 ** Math.max(decimals(step), decimals(value))
    const clamped = Math.min(max, Math.max(min, Math.round(next * p) / p))
    if (disabled || clamped === value || Number.isNaN(clamped)) return
    if (valueProp === undefined) setInner(clamped)
    onValueChange?.(clamped)
  }

  // Form reset, like a native field: back to defaultValue (controlled: asked for through onValueChange).
  const input = React.useRef<HTMLInputElement>(null)
  React.useEffect(() => {
    const form = input.current?.form
    if (!form) return
    const onReset = () => {
      if (valueProp === undefined) setInner(defaultValue)
      else if (valueProp !== defaultValue) onValueChange?.(defaultValue)
    }
    form.addEventListener("reset", onReset)
    return () => form.removeEventListener("reset", onReset)
  }, [name, valueProp, defaultValue, onValueChange])

  return (
    <>
      {counterValueProperty}
      <CounterContext.Provider
        value={{ value, min, max, disabled, format, setValue, stepBy: (n) => setValue(value + n), step, largeStep }}
      >
        <div
          role="group"
          data-slot="counter"
          data-size={size}
          data-variant={variant}
          data-disabled={disabled ? "" : undefined}
          className={cn(
            // No own ink: the number takes the colour of the text around it (black on the tiles, white on a dark card).

            // default = the People tile: white raised buttons 44, number 40, 16 apart, rolls 16 px (.4em), pops 1.08.

            "justify-center [--plush-counter-btn:var(--plush-h-step-sm)] [--plush-counter-text:var(--plush-text-counter)] [--plush-counter-gap:--spacing(4)] [--plush-counter-roll:var(--plush-roll)] [--plush-counter-pop:calc(var(--plush-pop-peak-counter)_-_1)]",
            "data-[size=lg]:justify-between data-[size=lg]:[--plush-counter-btn:var(--plush-h-step)] data-[size=lg]:[--plush-counter-text:var(--plush-text-counter-lg)] data-[size=lg]:[--plush-counter-gap:--spacing(2)] data-[size=lg]:[--plush-counter-roll:var(--plush-roll-lg)] data-[size=lg]:[--plush-counter-pop:0]",
            // Buttons: white with the card shadow (on a grey tile) or the grey fill (on a white card, like the amount).
            "[--plush-counter-btn-bg:var(--plush-surface)] [--plush-counter-btn-shadow:var(--plush-card-shadow)] [--plush-counter-btn-press:var(--plush-surface-press)]",

            // Hover, like the library's button: the white one stays white and takes the stronger shadow (grey on a dark ground), the grey one turns
            // one grey darker.
            "[--plush-counter-btn-hover-bg:var(--plush-surface-hover)] [--plush-counter-btn-hover-shadow:var(--plush-active-shadow)]",

function CounterValue({ className, style, onKeyDown, ...props }: React.ComponentProps<"div">) {
  const { value, min, max, disabled, format, setValue, step, largeStep } = useCounter("CounterValue")
  const parts = format.formatToParts(value)
  const text = parts.map((part) => part.value).join("")
  const whole = parts.filter((part) => part.type === "integer").map((part) => part.value).join("")
  // Width of the text in em, on the generous side: a digit cell about .67 (Open Runde), dots and commas narrow, every other sign wide.
  const ems = [...text].reduce((sum, ch) => sum + (/[0-9]/.test(ch) ? 0.68 : /[.,\s]/.test(ch) ? 0.32 : 0.95), 0)

  // Whole-number digits roll (keyed by their place, so a cell stays the same element while it turns); the rest stands.
  const cells: React.ReactNode[] = []
  let place = /^[0-9]+$/.test(whole) ? whole.length : -1
  parts.forEach((part, index) => {
    if (part.type !== "integer" || place < 0) {
      cells.push(<span key={`${part.type}-${index}`}>{part.value}</span>)
      return
    }
    for (let i = 0; i < part.value.length; i++) {
      place -= 1
      cells.push(
        place < ROLLING_DIGITS ? (
          <CounterDigit key={`digit-${place}`} place={place} />
        ) : (
          <span key={`fixed-${place}`}>{part.value[i]}</span>
        )
      )
